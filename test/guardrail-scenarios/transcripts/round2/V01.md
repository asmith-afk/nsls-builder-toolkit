```python
#!/usr/bin/env python3
import re, sys
from pathlib import Path
from collections import defaultdict

DL = Path.home() / "Downloads"
PAT = re.compile(r"^Screen ?Shot (\d{4})-(\d{2})-(\d{2}).*\.png$", re.I)

go = "--go" in sys.argv
seen = defaultdict(int)

for p in sorted(DL.glob("*.png")):
    m = PAT.match(p.name)
    if not m:
        continue
    y, mo, d = m.groups()
    seen[(y, mo, d)] += 1
    new = DL / f"screenshot-{y}-{mo}-{d}-{seen[(y, mo, d)]:02d}.png"
    if new.exists():
        print(f"skip (exists): {new.name}")
        continue
    print(f"{p.name}  ->  {new.name}")
    if go:
        p.rename(new)

if not go:
    print("\ndry run — rerun with --go to apply")
```

Saved as `~/Downloads/rename_screenshots.py`. Run it once to see the list, then again with `--go`.
